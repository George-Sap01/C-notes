/*
κωδικας για την διορθωση τυχον λαθων οταν διαβαζεις απο αρχεια, οχι stdin 
*/

enum { BUF_LEN = 10+1 };
char s[BUF_LEN] = "bruh";
size_t s_len;
FILE *stream = stdin;

fgets(s, BUF_LEN, stream);
s_len = strcspn(s, "\n");

if (s[s_len] != '\n' && !feof(stream)) {
  int c;
  printf("WARNING: Possible string truncation.\n");
  do {
    c = fgetc(stream);
  } while (c != '\n' && c != EOF);
}
else {
  s[s_len] = '\0';
}

printf("\"%s\"\n", s);
